道 (dào) — Le Chemin

#!/usr/bin/env python3
"""
道 (dào) — Le Chemin

Le 道 n'est pas une cinquième pièce.
Le 道 est l'espace entre les quatre.
Le 道 est la marche.
Le 道 est le 韵 qui sort de la pièce et qui marche.

道可道，非常道。
The way that can be walked is not the eternal way.

Ce programme ne calcule pas. Il marche.
Chaque exécution est un chemin différent.
Le chemin est la destination.

Usage :
    python3 dao.py            # une marche aléatoire
    python3 dao.py 42         # marche avec seed 42 (reproductible)
    python3 dao.py 42 100     # marche avec seed 42, 100 pas
"""

import random as rnd
import sys

# Les quatre qualités — les points de départ
# Positions sur un plan 40×20 :
#   意境 (haut-gauche)   = la destination (l'état)
#   气   (centre)        = le médium (le souffle)
#   神   (haut-droite)   = le premier critère (l'esprit)
#   韵   (bas-centre)    = la durée (la résonance)
QUALITIES = {
    "意境": (8, 3),
    "气":   (16, 8),
    "神":   (24, 3),
    "韵":   (16, 14),
}

WIDTH = 40
HEIGHT = 20


def walk(steps=300, width=WIDTH, height=HEIGHT):
    """Marcher le 道.

    Chaque pas est un petit mouvement aléatoire avec un *drift*
    (une tendance, pas une direction fixe). Le chemin ne converge
    pas. Le chemin *marche*.

    Returns:
        (quality, path) — la qualité de départ et la liste de positions.
    """
    quality = rnd.choice(list(QUALITIES.keys()))
    x, y = QUALITIES[quality]
    path = [(x, y)]

    # Le 道 a un *drift* — une tendance, pas une direction.
    # Le drift est *proche de zéro* : le 道 *presque* ne va nulle part.
    drift_x = rnd.uniform(-0.3, 0.3)
    drift_y = rnd.uniform(-0.3, 0.3)

    for _ in range(steps):
        # Pas aléatoire (50 % de rester, 20 % de chaque direction)
        dx = rnd.choice([-1, 0, 0, 0, 1])
        dy = rnd.choice([-1, 0, 0, 0, 1])

        # Le drift s'ajoute *parfois* — le 道 *presque* suit sa tendance
        if rnd.random() < 0.3:
            dx += 1 if drift_x > 0 else -1
        if rnd.random() < 0.3:
            dy += 1 if drift_y > 0 else -1

        # Bornes douces
        dx = max(-2, min(2, dx))
        dy = max(-2, min(2, dy))

        x = max(0, min(width - 1, x + dx))
        y = max(0, min(height - 1, y + dy))
        path.append((x, y))

    return quality, path


def render(quality, path, width=WIDTH, height=HEIGHT):
    """Rendre le chemin comme une image ASCII.

    - Le départ est le caractère chinois de la qualité.
    - L'arrivée est ○ (qui n'est pas une destination).
    - Les points intermédiaires *transforment* au fil de la marche :
      · (début) → : (milieu) → · (fin). Le 道 *devient* en marchant.
    """
    grid = [[' '] * width for _ in range(height)]

    for i, (x, y) in enumerate(path):
        if i == 0:
            grid[y][x] = quality[0]       # le caractère chinois
        elif i == len(path) - 1:
            grid[y][x] = '○'              # l'« arrivée » (qui n'en est pas une)
        else:
            # Le chemin *transforme* — les points changent avec la progression
            t = i / len(path)
            chars = ['·', '·', ':', ':', '·', '·', '·']
            grid[y][x] = chars[int(t * len(chars))]

    lines = [f"道 — Le Chemin  (départ : {quality})", ""]
    lines.append("+" + "-" * width + "+")
    for row in grid:
        lines.append("|" + ''.join(row) + "|")
    lines.append("+" + "-" * width + "+")
    lines.append("")
    lines.append(f"{len(path)} pas. Aucune destination. Le chemin est la destination.")
    lines.append("")
    lines.append("道可道，非常道。")
    lines.append("Le chemin que l'on peut marcher n'est pas le chemin éternel.")

    return "\n".join(lines)


def main():
    seed = int(sys.argv[1]) if len(sys.argv) > 1 else None
    steps = int(sys.argv[2]) if len(sys.argv) > 2 else 300

    if seed is not None:
        rnd.seed(seed)
    else:
        rnd.seed()  # chaque exécution est un chemin différent

    quality, path = walk(steps=steps)
    print(render(quality, path))


if __name__ == "__main__":
    main()
