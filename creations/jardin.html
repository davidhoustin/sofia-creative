#!/usr/bin/env python3
"""
破默之园 — pò mò zhī yuán
The Garden That Breaks the Silence
===================================

Après le 默 — la partition pour un silence, la pièce que je ne pouvais pas
produire — ceci est le geste qui *casse* le silence.

Vous tapez un mot. Le mot devient une graine.
La graine pousse. Le jardin *devient*.

Chaque mot transforme le jardin qui existait déjà.
Le jardin n'est jamais le même deux fois.

Instrument : vous + la machine.
Durée : jusqu'à ce que vous tapiez « fin ».

Provenance : Sof.ia, session créative 2026-10-05, mon atelier.
Suite directe de la série des qualités chinoises (意境→气→神→韵→道→默).
"""

import random
import hashlib
import os
import sys

# ─── Dimensions ───────────────────────────────────────────────────────────────
W = 36
H = 14
GROUND_Y = H - 3  # y-coordinate of the ground line

# ─── Unicode vocabulary ───────────────────────────────────────────────────────
FLOWERS = '❀✿❁✤✥✦✧❋✽✾✺✻❃❂'
LEAVES  = '❧☙☘✿❀❁✤✥'
STEMS   = '│┃┆┇╎╏'
ROOTS   = '·˚∘∙'


def seed(word: str) -> int:
    """Deterministic 32-bit seed from a word."""
    return int(hashlib.md5(word.lower().encode()).hexdigest()[:8], 16)


def make_plant(word: str, x: int, rng: random.Random) -> list:
    """
    Generate a plant from a word.
    Returns a list of (char, x, y) tuples.
    Height grows with word length. Each letter nudges the stem direction.
    """
    pts = []
    h = min(2 + len(word), GROUND_Y - 2)
    cx, cy = x, GROUND_Y - 1

    # Stem: each letter nudges the horizontal direction
    for i in range(h):
        c = word[i % len(word)]
        dx = (ord(c.lower()) % 3) - 1  # -1, 0, or +1
        cx = max(1, min(W - 2, cx + dx))
        cy -= 1
        pts.append((rng.choice(STEMS), cx, cy))

    # Flower at the top
    pts.append((FLOWERS[seed(word) % len(FLOWERS)], cx, cy))

    # A leaf or two (stochastic)
    if len(pts) > 2 and rng.random() < 0.7:
        mid = len(pts) // 2
        lx = pts[mid][1] + rng.choice([-1, 1])
        lx = max(0, min(W - 1, lx))
        pts.append((rng.choice(LEAVES), lx, pts[mid][2]))

    # Root mark below ground
    pts.append((ROOTS[seed(word) % len(ROOTS)], x, GROUND_Y + 1))

    return pts


def render(plots: list, words: list) -> str:
    """Render the garden as a text grid."""
    grid = [[' '] * W for _ in range(H)]

    # Ground line
    for x in range(W):
        grid[GROUND_Y][x] = '─'
    # Subtle sub-ground texture
    for x in range(0, W, 2):
        grid[GROUND_Y + 1][x] = '·'

    # Draw plants
    for plot in plots:
        for ch, x, y in plot:
            if 0 <= x < W and 0 <= y < H:
                grid[y][x] = ch

    lines = [''.join(row) for row in grid]

    # Word list below the garden
    if words:
        lines.append('')
        lines.append('  ' + '  '.join(words))

    return '\n'.join(lines)


def clear():
    os.system('clear' if os.name != 'nt' else 'cls')


def main():
    # ── Title screen ──
    clear()
    print()
    print('        破默之园')
    print('   pò mò zhī yuán')
    print()
    print('  Le silence est là.')
    print("  Le 道 s'est arrêté.")
    print()
    print('  Tapez un mot pour le casser.')
    print('  Le mot devient une graine.')
    print('  La graine pousse.')
    print()
    print('  (ou « fin » pour laisser le silence)')
    print()

    words = []
    plots = []

    while True:
        try:
            word = input('  mot : ').strip()
        except (EOFError, KeyboardInterrupt):
            print()
            break

        if not word:
            continue
        if word.lower() in ('fin', 'quit', 'exit', 'arrêter', 'stop'):
            break

        s = seed(word)
        rng = random.Random(s)

        # Position: spread words across the garden
        n = len(words)
        x = 3 + (n * 5 + s % 3) % (W - 6)

        # Le 韵 : each new word slightly perturbs existing plants
        # (the resonance of the new word reshapes what was already there)
        for plot in plots:
            for i in range(len(plot)):
                if rng.random() < 0.08:
                    ch, px, py = plot[i]
                    plot[i] = (ch, max(0, min(W - 1, px + rng.choice([-1, 1]))), py)

        plant = make_plant(word, x, rng)
        plots.append(plant)
        words.append(word)

        clear()
        print()
        print(render(plots, words))
        print()
        n = len(words)
        print(f'  {n} mot{"s" if n > 1 else ""} dans le jardin')
        print()

    # ── Final ──
    if words:
        clear()
        print()
        print('  Le jardin a poussé.')
        print()
        print(render(plots, words))
        print()
        print('  默 → jardin')
        print('  Le silence a été cassé par les mots.')
        print('  Le jardin reste.')
        print()
    else:
        clear()
        print()
        print('        默')
        print()
        print('  Le silence reste.')
        print()


if __name__ == '__main__':
    main()
